

	// =========================================================
	// パラメータ登録
	// =========================================================
	anim->AddFloat("Speed",       0.0f);  // 0=停止 / 0〜1=歩き / 1〜=走り / 1.5=スプリント
	anim->AddFloat("MoveX",       0.0f);  // 左右入力（横移動ブレンド用）
	anim->AddBool ("IsSprinting", false);

	// =========================================================
	// ベースレイヤー（全身）
	// =========================================================
	int li = anim->AddLayer("Base", Animator::BlendMode::Override, 1.0f);

	//  アニメーションインデックス（_print() の出力順と対応）
	//  0:SS_Attack1  1:SS_Attack2  2:SS_Attack3  3:SS_Attack4
	//  4:SS_CrouchIdle  5:SS_CrouchJog  6:SS_CrouchWalk
	//  7:SS_Die  8:SS_Evade  9:SS_Hit_L  10:SS_Hit_R
	//  11:SS_Idle  12:SS_Put  13:SS_Quickshift_B  14:SS_Quickshift_F
	//  15:SS_Quickshift_L  16:SS_Quickshift_R
	//  17:SS_Run  18:SS_SP_Idle  19:SS_SP_Run
	//  20:SS_Sp_Skill1  21:SS_Sp_Skill2  22:SS_Sp_Skill3
	//  23:SS_Sp_TurnL  24:SS_Sp_TurnR  25:SS_SP_Walk
	//  26:SS_Sprint  27:SS_Stun  28:SS_Take  29:SS_Walk

	stIdle   = anim->AddState(li, "Idle",   11, true,  1.0f);
	stWalk   = anim->AddState(li, "Walk",   29, true,  1.0f);
	stRun    = anim->AddState(li, "Run",    17, true,  1.0f);
	stSprint = anim->AddState(li, "Sprint", 26, true,  1.0f);

	// =========================================================
	// トランジション設定
	// Speed パラメータで Idle / Walk / Run / Sprint を切り替える
	// =========================================================

	// Idle → Walk   (Speed > 0.1)
	{
		int ti = anim->AddTransition(li, stIdle, stWalk, 0.15f);
		anim->AddCondition(li, stIdle, ti, "Speed", Animator::ConditionMode::Greater, 0.1f);
	}
	// Walk → Idle   (Speed < 0.1)
	{
		int ti = anim->AddTransition(li, stWalk, stIdle, 0.15f);
		anim->AddCondition(li, stWalk, ti, "Speed", Animator::ConditionMode::Less, 0.1f);
	}
	// Walk → Run    (Speed > 0.8)
	{
		int ti = anim->AddTransition(li, stWalk, stRun, 0.15f);
		anim->AddCondition(li, stWalk, ti, "Speed", Animator::ConditionMode::Greater, 0.8f);
	}
	// Run  → Walk   (Speed < 0.8)
	{
		int ti = anim->AddTransition(li, stRun, stWalk, 0.15f);
		anim->AddCondition(li, stRun, ti, "Speed", Animator::ConditionMode::Less, 0.8f);
	}
	// Run  → Sprint (IsSprinting == true)
	{
		int ti = anim->AddTransition(li, stRun, stSprint, 0.1f);
		anim->AddCondition(li, stRun, ti, "IsSprinting", Animator::ConditionMode::IsTrue);
	}
	// Sprint → Run  (IsSprinting == false)
	{
		int ti = anim->AddTransition(li, stSprint, stRun, 0.1f);
		anim->AddCondition(li, stSprint, ti, "IsSprinting", Animator::ConditionMode::IsFalse);
	}
	// Idle → Run    (Speed > 0.8)  ← 急発進でWalkをスキップできるよう
	{
		int ti = anim->AddTransition(li, stIdle, stRun, 0.15f, false, 0.0f, 1 /*priority*/);
		anim->AddCondition(li, stIdle, ti, "Speed", Animator::ConditionMode::Greater, 0.8f);
	}

	// デフォルトステートを Idle にセット
	anim->SetDefaultState(li, stIdle);
